# v0.91a 涌现问题宏观侵蚀接线 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 把 EmergentProblem 的活跃问题按等级持续侵蚀 Legitimacy / PublicOpinion / GovernanceResponseSystem（预算回补+冷却折扣），消退时经 ProblemResolvedEvent 一次性正向反馈。

**架构：** 三消费者在各自 Tick/Update 中直查 `EmergentProblemSystem.ActiveProblems` 快照（ServiceResolver 空安全解析），等级→反应量映射全部提为 internal static 纯函数；消退反馈集中在静态 helper `EmergentProblemWireup.HandleResolved`，由 GovernanceModule 订阅 `ProblemResolvedEvent`。

**技术栈：** C# / ONI mod 主程序集（net48）+ xUnit 测试项目（net8，`ONIModPack.Tests`，`InternalsVisibleTo` 已配置）。

**上游规格：** `docs/superpowers/specs/2026-08-27-politics-v0.91-emergentproblem-design.md`（第一~八节；生成机制为 v0.91b 计划）。

**常量默认值（规格第二节）：** Political/Crisis 磨损 0.002/0.005（合法性）、0.0005/0.001（民望）、消退回升 0.01/0.01、冷却折扣 Common/Political/Crisis = 0.8/0.6/0.5、预算回补 Political/Crisis = 0.02/0.05。

**命令约定：** 主构建 `dotnet build ONIModPack.csproj -c Debug`；测试 `dotnet test ONIModPack.Tests/ONIModPack.Tests.csproj -c Debug`（测试引用 `..\bin\Debug\net48\ONIModPack.dll`，**必须先 build 主项目再 test**）。

        /// Common 及以下不磨损。纯函数，供单测。

        internal static float GetEmergentWearRatePerSec(IEnumerable<EmergentProblem> problems, ReliefConfig config)
        {
            if (problems == null || config == null) return 0f;
            float rate = 0f;
            foreach (var p in problems)

                switch (p.Level)
                {
                    case ProblemLevel.Political: rate += config.ProblemLegitimacyWearPolitical; break;
                    case ProblemLevel.Crisis: rate += config.ProblemLegitimacyWearCrisis; break;
                }
            }
            return rate;
        }

            {
                if (p == null || !p.IsActive) continue;
                if (p.Level == ProblemLevel.Crisis) { hasCrisis = true; break; }
                if (p.Level == ProblemLevel.Political) hasPolitical = true;
            }
            if (hasCrisis) return config.ProblemBudgetRegenCrisis;
            if (hasPolitical) return config.ProblemBudgetRegenPolitical;
            return 0f;
        }

        /// <summary>v0.91：指定轴上活跃涌现问题最高等级 → 同轴冷却折扣系数（1.0=无折扣）。纯函数。</summary>
        internal static float GetAxisCooldownDiscount(IEnumerable<EmergentProblem> problems, SituationAxis axis, ReliefConfig config)
        {
            if (problems == null || config == null) return 1f;
            ProblemLevel max = ProblemLevel.Local;
            foreach (var p in problems)
            {
                if (p == null || !p.IsActive || p.SourceAxis != axis) continue;
                if (p.Level > max) max = p.Level;
            }
            switch (max)
            {
                case ProblemLevel.Common: return config.ProblemCooldownDiscountCommon;
                case ProblemLevel.Political: return config.ProblemCooldownDiscountPolitical;
                case ProblemLevel.Crisis: return config.ProblemCooldownDiscountCrisis;
                default: return 1f;
            }
        }

        /// <summary>v0.91：问题消退 → 清除该轴冷却相关累加（同轴边际计数归零、上次缓解时间移除）。</summary>
        public void ClearAxisCooldownAccum(SituationAxis axis)
        {
            _axisRecentCount[axis] = 0f;
            _axisLastReliefTime.Remove(axis);
        }
```

        /// <summary>v0.91：每结算间隔的涌现侵蚀（负值；key=话题）。纯函数。</summary>
        internal static Dictionary<OpinionTopic, float> GetEmergentOpinionErosion(
            IEnumerable<EmergentProblem> problems, ReliefConfig reliefConfig, float intervalSeconds)
        {
            var result = new Dictionary<OpinionTopic, float>();
            if (problems == null || reliefConfig == null) return result;

            ProblemLevel globalMax = ProblemLevel.Local;
            foreach (var p in problems)
            {
                if (p == null || !p.IsActive) continue;
                float rate;
                if (p.Level == ProblemLevel.Political) rate = reliefConfig.ProblemOpinionWearPolitical;
                else if (p.Level == ProblemLevel.Crisis) rate = reliefConfig.ProblemOpinionWearCrisis;
                else continue;

                if (p.Level > globalMax) globalMax = p.Level;
                var topic = PressureDemand.MapAxisToTopic(p.SourceAxis);
                float erosion = -rate * intervalSeconds;
                if (result.TryGetValue(topic, out var cur) && cur < erosion) continue; // 同话题保留更负者
                result[topic] = erosion;
            }

            // Law 话题按全局最高等级侵蚀（执政观感），不依赖具体轴
            if (globalMax >= ProblemLevel.Political)
            {
                float globalRate = globalMax == ProblemLevel.Crisis
                    ? reliefConfig.ProblemOpinionWearCrisis
                    : reliefConfig.ProblemOpinionWearPolitical;
                float globalErosion = -globalRate * intervalSeconds;
                if (!result.TryGetValue(OpinionTopic.Law, out var lawCur) || globalErosion < lawCur)
                    result[OpinionTopic.Law] = globalErosion;
            }
            return result;
        }

        internal void RefreshEmergentErosion(IEnumerable<EmergentProblem> problems)
        {
            _emergentErosion = GetEmergentOpinionErosion(problems, _reliefConfig,
                _config != null ? _config.UpdateInterval : 60f);
        }

        private float GetErosionFor(OpinionTopic topic)
            => _emergentErosion.TryGetValue(topic, out var v) ? v : 0f;
```

    /// <summary>
    /// v0.91 宏观接线静态助手：ProblemResolvedEvent 一次性正向反馈分发。
    /// 合法性回升 → Legitimacy.ApplyChange；民望回升 → IPublicOpinionSystem.ModifyOpinion（按轴映射话题）；
    /// 治理冷却清理 → GovernanceResponseSystem.ClearAxisCooldownAccum。
    /// 模块负责订阅，本助手负责分发（可单测，不依赖模块生命周期）。

    internal static class EmergentProblemWireup
    {
        public static SubscriptionToken Subscribe() =>
            EventBus.Subscribe<ProblemResolvedEvent>(HandleResolved);

        public static void HandleResolved(ProblemResolvedEvent e)
        {
            if (e == null) return;

            var config = ServiceResolver.OptionalService<ReliefConfig>();
            float legitRise = config != null ? config.ProblemResolvedLegitimacyRise : 0.01f;
            float opinionRise = config != null ? config.ProblemResolvedOpinionRise : 0.01f;

            ServiceResolver.OptionalService<Legitimacy>()?.ApplyChange(legitRise);

            var opinion = ServiceResolver.OptionalService<IPublicOpinionSystem>();
            opinion?.ModifyOpinion(PressureDemand.MapAxisToTopic(e.SourceAxis), opinionRise);

            ServiceResolver.OptionalService<GovernanceResponseSystem>()?.ClearAxisCooldownAccum(e.SourceAxis);
        }
    }
}
