# v0.91b 涌现法律生成机制 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 实现"涌现问题 → Sponsor 组织 → 简化管线审批 → 涌现法律成法（带溯源字段）"完整因果链，成法复用冻结事件 `LawEnactedEvent`。

**架构：** `EmergentLawGenerator`（5s tick 直查 `EmergentProblemSystem.ActiveProblems`，候选积累/去重/冷却/容量/轴→模板/参数调制/谈判）→ `EmergentLawPipeline`（Propose → Debating 30s 兼谈判 → Voting 简化表决 → Enacted/Rejected）；`PoliticalGroupFactory` 按 `axis:group` 建/复用 Sponsor；`Legislation` 新增 `SourceProblemId`/`SponsorGroupId` 溯源（含序列化）。

**技术栈：** C# / ONI mod 主程序集（net48）+ xUnit（net8，`ONIModPack.Tests`）。`InternalsVisibleTo("ONIModPack.Tests")` 已配置。

**上游规格：** `docs/superpowers/specs/2026-08-27-politics-v0.91-emergentproblem-design.md` 第九节；SPEC §3/§5 冻结条款（Legislation 仅新增字段、PoliticalGroup 仅赋值）。

**关键默认常量（规格 9.1/9.6/9.8）：** Common 候选 ≥60s；Political 提交 ≥120s；Crisis 驳回重提 180s；全局提交冷却 120s；Voting 阈值 0.5；执政偏置 +0.15；PoliticalLaw 容量 8；调制系数 k = 1+(Severity-0.5)×0.5；谈判力度 t = 0.5+Severity×0.5。

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;
using ONIModPack.Core;
using ONIModPack.Core.Services;
using ONIModPack.Content.SocialDynamics.Emergence;

namespace ONIModPack.Content.SocialDynamics.Politics
{
    /// <summary>
    /// v0.91 涌现 Sponsor 工厂：按「问题轴:影响群体」键建/复用政治组织。
    /// 同键组织殖民地生命周期内只建一次，复用累计组织度（+0.05/次）。
    /// PoliticalGroup 冻结字段只赋值不改名（SPEC §3）。
    /// </summary>
    public class PoliticalGroupFactory : IService
    {
        private readonly Dictionary<string, PoliticalGroup> _groups = new Dictionary<string, PoliticalGroup>();

        public bool IsInitialized { get; private set; }
        public IReadOnlyCollection<PoliticalGroup> Groups => _groups.Values;

        public void Initialize() { IsInitialized = true; }
        public void Shutdown() { _groups.Clear(); IsInitialized = false; }

        public PoliticalGroup GetOrCreate(SituationAxis axis, string groupLabel, float severity, string sourceIssue)
        {
            var key = axis + ":" + groupLabel;
            if (_groups.TryGetValue(key, out var existing))
            {
                existing.Organization = Mathf.Clamp01(existing.Organization + 0.05f);
                return existing;
            }

            var group = new PoliticalGroup
            {
                Id = Guid.NewGuid().ToString("N"),
                Name = GetAxisName(axis) + "·" + (groupLabel ?? "群体"),
                SourceIssue = sourceIssue,
                Organization = Mathf.Clamp01(0.1f + severity * 0.2f),
            };
            group.Goals.Add("解决" + GetAxisName(axis) + "问题");
            _groups[key] = group;
            return group;
        }

        private static string GetAxisName(SituationAxis axis) => axis switch
        {
            SituationAxis.LaborTension => "劳资紧张",
            SituationAxis.SocialUnrest => "社会动荡",
            SituationAxis.IdeologicalSplit => "意识分裂",
            SituationAxis.MoralFatigue => "道德疲惫",
            _ => axis.ToString(),
        };
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using ONIModPack.Core;
using ONIModPack.Core.Services;
using ONIModPack.Content.Legislation.Model;
using ONIModPack.Content.SocialDynamics;
using ONIModPack.Content.SocialDynamics.Emergence;
using ONIModPack.Content.SocialDynamics.Politics;

namespace ONIModPack.Content.Legislation.Runtime
{
    /// <summary>
    /// v0.91 涌现法律生成器：活跃 EmergentProblem 达到 Level≥Political 且持续 ≥ EmergentSubmitPoliticalSeconds
    /// 时，经 PoliticalGroupFactory 生成/复用 Sponsor，选择轴对应模板、按 Severity 调制参数、按 Stakeholders 结算谈判，
    /// 组装 PoliticalLaw（SourceProblemId/SponsorGroupId 溯源）提交 EmergentLawPipeline。
    /// 双重防护：同问题一次、同轴单待审、全局冷却、PoliticalLaw 容量上限（超限转入待容量轮询）。
    /// </summary>
    public class EmergentLawGenerator : IService
    {
        private EmergentProblemSystem _problems;
        private LegislationRegistry _registry;
        private PoliticalGroupFactory _groups;
        private EmergenceConfig _config;
        private EmergentLawPipeline _pipeline;

        private readonly HashSet<string> _submittedProblemIds = new HashSet<string>();
        private float _lastSubmitTime = -999f;

        public bool IsInitialized { get; private set; }

        public void Initialize()
        {
            _problems = ServiceResolver.OptionalService<EmergentProblemSystem>();
            _registry = ServiceResolver.OptionalService<LegislationRegistry>();
            _groups = ServiceResolver.OptionalService<PoliticalGroupFactory>();
            _config = ServiceResolver.OptionalService<EmergenceConfig>();
            _pipeline = ServiceResolver.OptionalService<EmergentLawPipeline>();
            IsInitialized = true;
            ModLogger.Info("[EmergentLaw] Generator initialized");
        }

        public void Shutdown()
        {
            _submittedProblemIds.Clear();
            _lastSubmitTime = -999f;
            IsInitialized = false;
        }

        /// <summary>调度 tick（5s）。每 tick 至多提交一例（全局冷却约束）。</summary>
        public void Tick(float deltaTime)
        {
            if (_problems == null || _registry == null || _pipeline == null) return;
            float globalCd = _config != null ? _config.EmergentGlobalSubmitCooldownSeconds : 120f;
            if (Time.time - _lastSubmitTime < globalCd) return;

            foreach (var problem in _problems.ActiveProblems.Values)
            {
                if (problem == null || !problem.IsActive) continue;
                if (!ShouldSubmit(problem)) continue;
                if (TryGenerate(problem)) return; // 单 tick 单提交
            }
        }

        internal bool ShouldSubmit(EmergentProblem problem)
        {
            float submitSecs = _config != null ? _config.EmergentSubmitPoliticalSeconds : 120f;
            if (problem.Level < ProblemLevel.Political) return false;
            if (problem.Persistence < submitSecs) return false;
            if (_submittedProblemIds.Contains(problem.Id)) return false;     // 同问题去重
            if (_pipeline.HasPendingForAxis(problem.SourceAxis)) return false; // 同轴单待审
            if (PoliticalLawCount() >= Capacity) return false;               // 待容量

            if (_pipeline.RejectedAt.TryGetValue(problem.Id, out var rejectedAt))
            {
                float resub = _config != null ? _config.EmergentResubmitCooldownSeconds : 180f;
                if (Time.time - rejectedAt < resub) return false;            // Crisis 重提冷却
            }
            return true;
        }

        private int Capacity => _config != null ? _config.EmergentLawCapacity : 8;

        private int PoliticalLawCount()
        {
            int n = 0;
            foreach (var law in _registry.GetAllLaws())
                if (law.Level == LawLevel.PoliticalLaw && law.Status != LawStatus.Repealed) n++;
            return n;
        }

        internal bool TryGenerate(EmergentProblem problem)
        {
            var category = MapAxisToLawCategory(problem.SourceAxis);
            var template = _registry.GetAllTemplates().FirstOrDefault(t => t.Category == category);
            if (template == null)
            {
                ModLogger.Warning($"[EmergentLaw] No template for category {category} (axis {problem.SourceAxis})");
                return false;
            }

            string sponsorLabel = problem.AffectedGroups != null && problem.AffectedGroups.Length > 0
                ? problem.AffectedGroups[0] : "受影响群体";
            var sponsor = _groups?.GetOrCreate(problem.SourceAxis, sponsorLabel, problem.Severity, problem.Id);

            var paramValues = BuildParameters(template, problem.Severity);

            var law = new LawInstance
            {
                Id = "emerg_" + Guid.NewGuid().ToString("N").Substring(0, 8),
                Name = "【涌现】" + template.Name,
                TemplateId = template.Id,
                Status = LawStatus.Proposed,
                Level = LawLevel.PoliticalLaw,
                AuthorityCost = 0f,
                Scope = "全殖民地",
                EnforcementMode = EnforcementModel.AgencyRequired,
                SourceProblemId = problem.Id,
                SponsorGroupId = sponsor?.Id,
            };
            foreach (var kv in paramValues)
                law.Parameters[kv.Key] = new ParameterValue { ParameterId = kv.Key, Value = kv.Value };

            _pipeline.Submit(law, problem);
            _submittedProblemIds.Add(problem.Id);
            _lastSubmitTime = Time.time;
            ModLogger.Info($"[EmergentLaw] 提交涌现提案 '{law.Name}'（问题 {problem.Id} 轴 {problem.SourceAxis} 严重度 {problem.Severity:P0}）");
            return true;
        }

        internal static Dictionary<string, object> BuildParameters(LawTemplate template, float severity)
        {
            var values = new Dictionary<string, object>();
            foreach (var p in template.Parameters)
            {
                if (IsNumeric(p))
                {
                    float tuned = ModulateParameter(p.Default, severity);
                    values[p.Id] = Mathf.Clamp(tuned, p.Min, p.Max);
                }
                else
                {
                    values[p.Id] = p.Default;
                }
            }

            if (template.Stakeholders == null || template.Stakeholders.Count == 0) return values;

            foreach (var s in template.Stakeholders)
            {
                if (!values.TryGetValue(s.ParameterId, out var obj) || !(obj is float current)) continue;
                var p = template.Parameters.FirstOrDefault(x => x.Id == s.ParameterId);
                float negotiated = NegotiateParameter(current, s, severity);
                values[s.ParameterId] = p != null ? Mathf.Clamp(negotiated, p.Min, p.Max) : negotiated;
            }
            return values;
        }

        private static bool IsNumeric(LawParameter p)
            => p != null && (p.Type == LawParameterType.Integer || p.Type == LawParameterType.Float);

        // ---- 纯函数（可测） ----

        /// <summary>问题轴 → 模板类别（规格 9.3 映射表）。</summary>
        internal static LawCategory MapAxisToLawCategory(SituationAxis axis) => axis switch
        {
            SituationAxis.LaborTension => LawCategory.Labor,
            SituationAxis.SocialUnrest => LawCategory.Security,
            SituationAxis.IdeologicalSplit => LawCategory.Religion,
            SituationAxis.MoralFatigue => LawCategory.Welfare,
            _ => LawCategory.PublicService,
        };

        /// <summary>k = 1 + (severity - 0.5) × 0.5，k ∈ [0.75, 1.25]（规格 9.3）。</summary>
        internal static float ModulateParameter(float baseValue, float severity)
            => baseValue * (1f + (severity - 0.5f) * 0.5f);

        /// <summary>谈判结算（规格 9.5）：t = 0.5 + severity × 0.5；final = tuned × (1 + Direction × Amplitude × t)。</summary>
        internal static float NegotiateParameter(float tuned, LawStakeholder stakeholder, float severity)
        {
            if (stakeholder == null) return tuned;
            float t = 0.5f + severity * 0.5f;
            return tuned * (1f + stakeholder.Direction * stakeholder.Amplitude * t);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;
using ONIModPack.Core;
using ONIModPack.Core.Services;
using ONIModPack.Content.Legislation.Events;
using ONIModPack.Content.Legislation.Model;
using ONIModPack.Content.SocialDynamics;
using ONIModPack.Content.SocialDynamics.Emergence;
using ONIModPack.Content.SocialDynamics.Politics;

namespace ONIModPack.Content.Legislation.Runtime
{
    /// <summary>
    /// v0.91 简化涌现管线（与 LegislationManager 完整管线并行、零代码交集）：
    /// Propose → Debating(30s，谈判已在生成器结算) → Voting(简化表决 30s) → Enacted / Rejected。
    /// Enacted 注册入 LegislationRegistry 并发布冻结事件 LawEnactedEvent（GovernanceModule 既有接线自动泄压/合法性回升）；
    /// Rejected 发布 LawProposalStatusChangedEvent 并记录驳回时间（供 Crisis 重提）。
    /// </summary>
    public class EmergentLawPipeline : IService
    {
        private enum Stage { Debating, Voting }

        private class Entry
        {
            public LawInstance Law;
            public EmergentProblem Problem;
            public Stage Stage = Stage.Debating;
            public float EnteredAt;
        }

        public const float DebatingSeconds = 30f;
        public const float VotingSeconds = 30f;
        public const float RulingBias = 0.15f;

        private readonly List<Entry> _entries = new List<Entry>();
        private readonly Dictionary<string, float> _rejectedAt = new Dictionary<string, float>();
        private LegislationRegistry _registry;
        private IRulingFactionService _ruling;

        /// <summary>测试缝：&gt;=0 时作为当前时间（秒）并允许手动推进；&lt;0 时用 Time.time。</summary>
        public float DebugNow = -1f;

        public bool IsInitialized { get; private set; }
        public IReadOnlyDictionary<string, float> RejectedAt => _rejectedAt;

        public void Initialize()
        {
            _registry = ServiceResolver.OptionalService<LegislationRegistry>();
            _ruling = ServiceResolver.OptionalService<IRulingFactionService>();
            IsInitialized = true;
            ModLogger.Info("[EmergentLaw] Pipeline initialized");
        }

        public void Shutdown()
        {
            _entries.Clear();
            _rejectedAt.Clear();
            IsInitialized = false;
        }

        public void Submit(LawInstance law, EmergentProblem problem)
        {
            _entries.Add(new Entry { Law = law, Problem = problem, Stage = Stage.Debating, EnteredAt = Now() });
            EventBus.Publish(new LawProposalStatusChangedEvent { LawId = law.Id, Status = "Debating" });
        }

        public bool HasPendingForAxis(SituationAxis axis)
        {
            foreach (var e in _entries)
                if (e.Problem != null && e.Problem.SourceAxis == axis) return true;
            return false;
        }

        /// <summary>调度 tick（5s）。</summary>
        public void Tick(float deltaTime)
        {
            float now = Now();
            for (int i = _entries.Count - 1; i >= 0; i--)
            {
                var e = _entries[i];
                if (e.Stage == Stage.Debating && now - e.EnteredAt >= DebatingSeconds)
                {
                    e.Stage = Stage.Voting;
                    e.EnteredAt = now;
                    EventBus.Publish(new LawProposalStatusChangedEvent { LawId = e.Law.Id, Status = "Voting" });
                    continue;
                }
                if (e.Stage == Stage.Voting && now - e.EnteredAt >= VotingSeconds)
                {
                    Resolve(e);
                    _entries.RemoveAt(i);
                }
            }
        }

        private void Resolve(Entry e)
        {
            bool passed = ComputeSupport(e.Problem.AffectedGroups, e.Problem.Severity,
                RulingBiasFor(e.Law.TemplateId)) >= PassThreshold;

            if (passed && TryRulingVeto(e.Law.TemplateId))
            {
                passed = false; // 执政一票否决（SPEC 1.2 表决期 VetoLaw 语义）
            }

            if (passed)
            {
                e.Law.Status = LawStatus.Enacted;
                _registry?.RegisterLaw(e.Law);
                var template = _registry?.GetTemplate(e.Law.TemplateId);
                EventBus.Publish(new LawEnactedEvent
                {
                    LawId = e.Law.Id,
                    LawName = e.Law.Name,
                    TemplateId = e.Law.TemplateId,
                    Category = template != null ? template.Category : LawCategory.PublicService,
                });
                ModLogger.Info($"[EmergentLaw] 涌现法律通过 '{e.Law.Name}'（{e.Law.Id}，溯源问题 {e.Law.SourceProblemId}）");
            }
            else
            {
                e.Law.Status = LawStatus.Rejected;
                _rejectedAt[e.Problem.Id] = Now();
                EventBus.Publish(new LawProposalStatusChangedEvent { LawId = e.Law.Id, Status = "Rejected" });
                ModLogger.Info($"[EmergentLaw] 涌现提案被驳回 '{e.Law.Name}'（问题 {e.Problem.Id}）");
            }
        }

        private float PassThreshold
        {
            get
            {
                var cfg = ServiceResolver.OptionalService<EmergenceConfig>();
                return cfg != null ? cfg.EmergentLawPassThreshold : 0.5f;
            }
        }

        private float RulingBiasFor(string templateId)
        {
            if (_ruling == null || _registry == null) return 0f;
            var template = _registry.GetTemplate(templateId);
            if (template == null) return 0f;
            return _ruling.GetProposalBonus(template.Category) < 0f ? RulingBias : 0f; // 执政偏好类别享红利
        }

        private bool TryRulingVeto(string templateId)
        {
            if (_ruling == null || !_ruling.CanVeto || _registry == null) return false;
            var template = _registry.GetTemplate(templateId);
            if (template == null) return false;
            if (_ruling.GetProposalBonus(template.Category) < 0f) return false; // 执政偏好类别不否决
            return _ruling.TryConsumeVeto(out _);
        }

        private float Now() => DebugNow >= 0f ? DebugNow : Time.time;

        /// <summary>简化表决支持度（规格 9.6）：恒等权群体下 = severity + rulingBias。纯函数。</summary>
        internal static float ComputeSupport(string[] affectedGroups, float severity, float rulingBias)
            => Mathf.Clamp01(severity + rulingBias);
    }
}
```
